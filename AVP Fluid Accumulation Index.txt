
--Identifying all volumetric inputevents that occurred in our analytic sample--
SELECT DISTINCT i.itemid, d.label
FROM `physionet-data.mimiciv_3_1_icu.inputevents` i
LEFT JOIN `physionet-data.mimiciv_3_1_icu.d_items` d
  ON i.itemid = d.itemid
WHERE i.stay_id IN(
    SELECT DISTINCT a.stay_id
        FROM `mimic-iv-ext-2025.avp.avp_sample` a)
          AND i.amountuom in ('mL', 'L','nL', 'pL', 'µL' ); --Excluding itemid 226361 "Pre-Admission/Non-ICU Intake" from our list--



--Identifying all volumetric outputevents that occurred in our analytic sample--
SELECT DISTINCT o.itemid, d.label, o.valueuom
FROM `physionet-data.mimiciv_3_1_icu.outputevents` o
LEFT JOIN `physionet-data.mimiciv_3_1_icu.d_items` d
  ON o.itemid = d.itemid
WHERE o.stay_id IN(
    SELECT DISTINCT a.stay_id
        FROM `mimic-iv-ext-2025.avp.avp_sample` a); --Excluding itemid 226633 "Pre-Admission" from our list--

--https://www.frontiersin.org/journals/medicine/articles/10.3389/fmed.2022.714384/full#supplementary-material--
--1)All inclusion/exclusion criteria were applied
--2)Pulled all inputevents item IDs that occurred in our analytic sample
--3)Pulled all outputevents item IDs that occurred in our analytic sample
--4)Modified the Dong et al. fluid balance code to use our population-specific input and output events

        p1.starttime, 
        p1.itemid,

        p1.ordercategoryname,
        p1.secondaryordercategoryname,
   round(case
      when p1.amountuom='L' then p1.amount*1000
                    when p1.amountuom='mL' then p1.amount
                    else NULL end) as amount
      from `physionet-data.mimiciv_3_1_icu.inputevents` p1 
       left join `mimic-iv-ext-2025.avp.avp_sample` p2
     on p1.stay_id = p2.stay_id 
     where p1.starttime between p2.avpstart and (DATETIME_ADD( p2.avpstart,interval '2' DAY))
),
t2 as --t2 is dialysis outputs within 2 days after avp initiation(chartevents)
(select p1.stay_id, p1.itemid, p1.valuenum
   from `physionet-data.mimiciv_3_1_icu.chartevents` p1
    left join `mimic-iv-ext-2025.avp.avp_sample` p2
      on p1.stay_id= p2.stay_id
        where p1.charttime between p2.avpstart and (DATETIME_ADD(p2.avpstart,interval '2' DAY)) 
),

--Merging our two fb tables together--
CREATE OR REPLACE TABLE `mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes` AS
SELECT t1.stay_id, t2.total_in_avpstart, t2.total_out_avpstart, t3.total_in_vasoactivestart, t3.total_out_vasoactivestart
FROM mimic-iv-ext-2025.avp.avp_sample t1
LEFT JOIN mimic-iv-ext-2025.avp.avp_sample_fb t2
  ON t1.stay_id = t2.stay_id
    LEFT JOIN mimic-iv-ext-2025.avp.avp_sample_fb_vasoactivestart t3
      ON t1.stay_id = t3.stay_id;



--Bringing in subject ID and date of death (captures out of hospital mortality)--
CREATE OR REPLACE TABLE mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes_dod AS
SELECT f.*, i.subject_id, i.hadm_id, p.dod
FROM mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes f
LEFT JOIN physionet-data.mimiciv_3_1_icu.icustays i
  ON f.stay_id = i.stay_id
    LEFT JOIN physionet-data.mimiciv_3_1_hosp.patients p 
      ON p.subject_id = i.subject_id;
SELECT *
FROM mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes_dod;


--Bringing in individual CCI components for propensity score model--
CREATE OR REPLACE TABLE mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes_dod_cci AS
SELECT a.*, b.age_score AS age_score_cci, b.myocardial_infarct AS myocardial_infarct_cci,
       b.peripheral_vascular_disease AS peripheral_vascular_disease_cci, b.cerebrovascular_disease AS        
      cerebrovascular_disease_cci, b.dementia AS dementia_cci, b.chronic_pulmonary_disease AS 
      chronic_pulmonary_disease_cci,
       b.rheumatic_disease AS rheumatic_disease_cci, b.peptic_ulcer_disease AS peptic_ulcer_disease_cci, 
       b.mild_liver_disease AS mild_liver_disease_cci, b.diabetes_without_cc AS diabetes_without_cc_cci,
       b.diabetes_with_cc AS diabetes_with_cc_cci, b.paraplegia AS paraplegia_cci, b.renal_disease AS renal_disease_cci,
       b.malignant_cancer AS malignant_cancer_cci, b.severe_liver_disease AS severe_liver_disease_cci,
       b.metastatic_solid_tumor AS metastatic_solid_tumor_cci, b.aids AS aids_cci
FROM mimic-iv-ext-2025.avp.avp_sample_fb_bothtimes_dod a
  LEFT JOIN physionet-data.mimiciv_3_1_derived.charlson b
    ON a.hadm_id = b.hadm_id;
